       ```python
       """LiveKit AgentSession factory wiring Deepgram STT, Cartesia TTS, Silero VAD,
       and our LangGraph-backed DailyLLMBridge.
       """
       import logging
       from typing import AsyncIterator

       from livekit.agents import llm as agents_llm
       from livekit.plugins import cartesia, deepgram, silero

       from daily.config import Settings
       from daily.worker.llm_bridge import DailyLLMBridge

       logger = logging.getLogger(__name__)


       class DailyAgentLLM(agents_llm.LLM):
           """Adapter exposing DailyLLMBridge as a livekit-agents LLM."""

           def __init__(self, bridge: DailyLLMBridge) -> None:
               super().__init__()
               self._bridge = bridge

           def chat(self, *, chat_ctx, fnc_ctx=None, **kwargs):
               # Extract the most recent user message
               user_msg = ""
               for msg in reversed(chat_ctx.messages):
                   if getattr(msg, "role", None) == "user":
                       user_msg = msg.content if isinstance(msg.content, str) else str(msg.content)
                       break
               return _DailyLLMStream(self, chat_ctx, self._bridge, user_msg)


       class _DailyLLMStream(agents_llm.LLMStream):
           def __init__(self, llm, chat_ctx, bridge: DailyLLMBridge, user_msg: str) -> None:
               super().__init__(llm=llm, chat_ctx=chat_ctx, fnc_ctx=None)
               self._bridge = bridge
               self._user_msg = user_msg

           async def _run(self) -> None:
               async for delta in self._bridge.stream_response(self._user_msg):
                   chunk = agents_llm.ChatChunk(
                       request_id="",
                       choices=[
                           agents_llm.Choice(
                               delta=agents_llm.ChoiceDelta(role="assistant", content=delta),
                               index=0,
                           )
                       ],
                   )
                   self._event_ch.send_nowait(chunk)


       def build_voice_pipeline(bridge: DailyLLMBridge, settings: Settings):
           """Construct an AgentSession ready to .start() against a JobContext.room."""
           from livekit.agents import AgentSession  # local import keeps test stubs simple

           return AgentSession(
               stt=deepgram.STT(model="nova-3", api_key=settings.deepgram_api_key),
               tts=cartesia.TTS(api_key=settings.cartesia_api_key),
               vad=silero.VAD.load(),
               llm=DailyAgentLLM(bridge),
           )
       ```
       Note: If Context7 docs reveal the LLMStream interface differs (e.g. ChatChunk fields are `id`/`choices` not `request_id`/`choices`, or `_event_ch` is named differently), update implementation accordingly. Pin to whatever the installed `livekit-agents` package exposes.

    ```python
    """LiveKit agent entrypoint — full voice loop over WebRTC.

    On dispatch:
      1. Connect to the room.
      2. Resolve user_id from room name (parse_user_id_from_room).
      3. Wait for the participant to publish their mic track.
      4. Load the user's session bundle (graph + briefing) via load_user_session_state.
      5. Build the AgentSession (STT/TTS/VAD/LLM via build_voice_pipeline).
      6. Start the session against ctx.room.
      7. Speak the briefing as the opening turn (allow_interruptions=True so user can barge in).
      8. AgentSession handles the rest of the voice loop until participant leaves.
    """
    import logging

    from livekit.agents import Agent, JobContext

    from daily.config import Settings
    from daily.worker.identity import parse_user_id_from_room
    from daily.worker.llm_bridge import DailyLLMBridge
    from daily.worker.state import load_user_session_state
    from daily.worker.voice_pipeline import build_voice_pipeline

    logger = logging.getLogger(__name__)

    _AGENT_INSTRUCTIONS = (
        "You are dAIly, a personal voice briefing assistant. Be concise and conversational. "
        "Respond in short, natural sentences suitable for spoken delivery."
    )


    async def entrypoint(ctx: JobContext) -> None:
        await ctx.connect()
        user_id = parse_user_id_from_room(ctx.room.name)
        if user_id is None:
            logger.warning("worker: invalid room name %r — disconnecting", ctx.room.name)
            await ctx.shutdown(reason="invalid-room-name")
            return

        logger.info("worker: dispatched room=%s user_id=%d", ctx.room.name, user_id)

        # Wait for the mobile client to publish its mic track before starting voice loop.
        await ctx.wait_for_participant()

        settings = Settings()
        async with load_user_session_state(user_id, settings) as bundle:
            bridge = DailyLLMBridge(
                graph=bundle.graph,
                config=bundle.config,
                initial_state=bundle.initial_state,
            )
            session = build_voice_pipeline(bridge, settings)

            await session.start(
                agent=Agent(instructions=_AGENT_INSTRUCTIONS),
                room=ctx.room,
            )

            # First turn: speak the briefing if we have one.
            briefing = bundle.briefing_narrative
            if briefing:
                logger.info("worker: speaking briefing (%d chars)", len(briefing))
                await session.say(briefing, allow_interruptions=True)
            else:
                logger.info("worker: no cached briefing for user_id=%d", user_id)
                await session.say(
                    "Good morning. I don't have a briefing cached yet — what would you like to talk about?",
                    allow_interruptions=True,
                )

            # AgentSession owns the voice loop from here. Block until the room ends.
            try:
                await ctx.wait_for_disconnect()
            except AttributeError:
                # Older API: just wait for the room to end via context shutdown
                await ctx.shutdown_event.wait()
    ```
    Note: `wait_for_participant`, `wait_for_disconnect`, and `shutdown_event` exact names depend on the installed livekit-agents version. Check via Context7. If `wait_for_disconnect` does not exist, fall back to `await asyncio.Event().wait()` guarded by `ctx.add_shutdown_callback`.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Mobile client → worker (audio in) | Mic audio crosses LiveKit WebRTC; worker receives via Deepgram plugin |
| Worker → mobile client (audio out) | TTS audio published as room track; client renders via LiveKit SDK |
| Worker → Deepgram/Cartesia APIs | Worker holds API keys; STT/TTS providers see audio + text |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-20.1-06 | Information Disclosure | Briefing narrative spoken before participant joined | mitigate | Use `await ctx.wait_for_participant()` before `session.say(briefing)` — confirms client is actually subscribed |
| T-20.1-07 | Spoofing | Stranger joins another user's room | mitigate (existing) | LiveKit room access is enforced by Phase 18 server-signed JWT; worker only validates room-name format, not access |
